# Spec 28.02 — Database Schema `grp`

File: `grp/store/migrations/0001_grp_schema.sql` (or an Alembic revision if the repo uses Alembic — match the repo). Applied by `make migrate` (28.11). All timestamps `timestamptz`, stored UTC. Postgres 16.

---

## 1. Extensions and fallbacks

```sql
do $$ begin
  begin execute 'create extension if not exists vector';  exception when others then raise notice 'pgvector unavailable'; end;
  begin execute 'create extension if not exists postgis'; exception when others then raise notice 'postgis unavailable'; end;
end $$;
```

`grp/store/repo.py` detects at startup:

- `has_pgvector = exists(select 1 from pg_extension where extname='vector')` → `raw_items.embedding vector(384)` with `<=>` cosine in SQL; else column is `real[]` and cosine is computed in Python over the dedup window only (28.03 §6).
- `has_postgis` → `notams.geom geometry(Polygon,4326)`; else `null` and `bbox jsonb` is used with shapely (28.04).

Local docker-compose uses image `pgvector/pgvector:pg16` so pgvector is present; PostGIS is optional in v1.

## 2. DDL

```sql
create schema if not exists grp;

-- ------------------------------------------------------------------ raw news / social items
create table grp.raw_items (
  id            bigserial primary key,
  source        text not null check (source in ('newsapi','worldnews','gdelt','manual','x_paste')),
  source_id     text not null,                     -- provider id or url hash
  published_at  timestamptz not null,
  fetched_at    timestamptz not null default now(),
  title         text,
  body          text,
  url           text,
  outlet        text,
  lang          text,
  author        text,                              -- X handle for x_paste
  embedding     vector(384),                       -- real[] if no pgvector
  event_id      bigint,                            -- set by dedup; FK added below
  classified    boolean not null default false,
  raw           jsonb not null default '{}'::jsonb,
  unique (source, source_id)
);
create index raw_items_published_idx on grp.raw_items (published_at desc);
create index raw_items_unclassified_idx on grp.raw_items (classified) where classified = false;

-- ------------------------------------------------------------------ canonical events
create table grp.events (
  id               bigserial primary key,
  theatre          text not null,
  first_seen       timestamptz not null,
  last_seen        timestamptz not null,
  actor            text not null,
  target           text,
  type             text not null check (type in
                     ('threat','strike','seizure','sanction','talks','walk_back','denial',
                      'airspace','aircraft','vessel','other')),
  severity         smallint not null check (severity between 1 and 5),
  source_tier      smallint not null check (source_tier between 1 and 3),
  confidence       real not null check (confidence between 0 and 1),
  summary          text not null,
  item_count       integer not null default 1,
  parent_threat_id bigint references grp.events(id) on delete set null,
  resolution       text check (resolution in ('follow_through','walk_back','unresolved')),
  resolved_at      timestamptz,
  meta             jsonb not null default '{}'::jsonb,
  created_at       timestamptz not null default now(),
  updated_at       timestamptz not null default now()
);
create index events_theatre_time_idx on grp.events (theatre, first_seen desc);
create index events_actor_type_idx  on grp.events (actor, type, first_seen desc);
create index events_open_threats_idx on grp.events (actor, first_seen) where type = 'threat' and resolution is null;
alter table grp.raw_items add constraint raw_items_event_fk foreign key (event_id) references grp.events(id) on delete set null;

-- ------------------------------------------------------------------ NOTAMs
create table grp.notams (
  id              text primary key,               -- FAA notam id
  number          text,
  fir             text not null,
  issued_at       timestamptz not null,
  effective_from  timestamptz,
  effective_to    timestamptz,                    -- null = PERM
  classification  text,
  text            text not null,
  keywords        text[] not null default '{}',
  severity        smallint not null check (severity between 1 and 5),
  geom            geometry(Polygon,4326),         -- null if no postgis
  bbox            jsonb,                          -- {minlon,minlat,maxlon,maxlat}
  event_id        bigint references grp.events(id) on delete set null,
  fetched_at      timestamptz not null default now(),
  raw             jsonb not null default '{}'::jsonb
);
create index notams_fir_eff_idx on grp.notams (fir, effective_to);

-- ------------------------------------------------------------------ aircraft
create table grp.aircraft_positions (
  icao24        text not null,
  ts            timestamptz not null,
  callsign      text,
  type_code     text,
  operator      text,
  origin_country text,
  lon           double precision,
  lat           double precision,
  baro_alt_m    real,
  velocity_ms   real,
  on_ground     boolean,
  geofence      text,
  primary key (icao24, ts)
);
create index aircraft_geofence_ts_idx on grp.aircraft_positions (geofence, ts desc) where geofence is not null;

create table grp.aircraft_landings (
  id          bigserial primary key,
  icao24      text not null,
  callsign    text,
  type_code   text,
  geofence    text not null,
  ts          timestamptz not null,

create table grp.airport_delays (
  airport        text not null,

-- ------------------------------------------------------------------ vessels

create table grp.vessel_positions (

create index vessel_pos_geofence_ts_idx on grp.vessel_positions (geofence, ts desc) where geofence is not null;

create table grp.vessel_anomalies (
  id        bigserial primary key,

-- ------------------------------------------------------------------ physical (EIA-derived)

-- ------------------------------------------------------------------ positioning

-- ------------------------------------------------------------------ scoring

  d               date not null,
  theatre         text not null,

create table grp.components (
  ts               timestamptz not null,
  theatre          text not null,

create table grp.scores (
  ts             timestamptz not null,
  theatre        text not null,
  grp            real not null,
  regime         text not null check (regime in ('calm','elevated','stressed','crisis')),
  fade           boolean not null,
  model_version  text not null,
  stale          text[] not null default '{}',

create index scores_theatre_ts_idx on grp.scores (theatre, ts desc);

-- ------------------------------------------------------------------ ops
create table grp.feed_status (
  feed                 text primary key,
```

## 3. Views (for Grafana and the API)

```sql
create or replace view grp.v_latest_score as
  select distinct on (theatre) * from grp.scores order by theatre, ts desc;

create or replace view grp.v_events_review as
  select e.id, e.first_seen, e.actor, e.target, e.type, e.severity, e.source_tier, e.confidence,
         e.summary, e.item_count, e.resolution,
         (select url from grp.raw_items r where r.event_id = e.id order by published_at limit 1) as first_url
  from grp.events e;

create or replace view grp.v_vessels_latest as
  select distinct on (p.mmsi) p.*, s.name, s.ship_type, s.destination, s.draught_m
  from grp.vessel_positions p left join grp.vessel_static s using (mmsi)
  where p.ts > now() - interval '6 hours'
  order by p.mmsi, p.ts desc;

create or replace view grp.v_feed_health as
  select feed, expected_cadence_min, last_success,
         extract(epoch from (now() - last_success))/60 as age_min,
         (extract(epoch from (now() - last_success))/60) > 2 * expected_cadence_min as stale,
         last_error, last_error_msg
  from grp.feed_status;
```

## 4. Retention (`make prune`, weekly cron later)

- `raw_items` older than 180 d with `event_id` set: keep `id, event_id, url, published_at`, null `body` and `embedding`.
- `vessel_positions` older than 90 d: delete (aggregates in `ais_daily` remain).
- `aircraft_positions` older than 90 d: delete (`aircraft_landings` remain).
- `components`, `scores`, `taco`, `positioning`, `physical_weekly`, `ais_daily`: keep forever.
